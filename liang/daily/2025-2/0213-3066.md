## 3066. Minimum Operations to Exceed Threshold Value II
### C++
``` c++
class Solution {
public:
    int minOperations(vector<int>& nums, int k) {
        // heapify make heap O(N)
        priority_queue<long long, vector<long long>, greater<long long>> q(nums.begin(), nums.end());
        bool isLarger = false;

        int count=0;
        while(q.size()>=1){
            long long n1 = q.top();
            q.pop();
            if (k<=n1)
                return count;

            long long n2 = q.top();
            q.pop();
            q.push(2*n1+n2); // min(x,y)*2 + max(x,y)
            count++;

        }

        return -1;
    }
};

// priority queue 會有機制使得 就算 vector 1,2,3,4,5,...由小到大
// heap 長相會自行調整成以下這樣
//           2
//        /    \
//       3      4
//      / \    /  \
//     7   6  5    8
//    /
//   9
// Heap 是 Complete binary tree
// 因此確保每次 insert or pop 都是 O(logN)
// Worse : O(NlogN)
// 因為頂多做 N-1 次 operation
// 又每次 pop , pop , push -> O(3*log(N))
```
