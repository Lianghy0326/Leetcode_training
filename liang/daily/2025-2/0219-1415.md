## 1415. The k-th Lexicographical String of All Happy Strings of Length n
### C++
``` c++
class Solution {
public:
    string getHappyString(int n, int k) {
        int num = 3*(1<<(n-1));
        if (k>num) return "";

        int tmp=0;
        int block_size, index;
        string ans="";
        array<char, 2> arr[3] = {{'b','c'},{'a','c'},{'a','b'}};
        
        while(tmp < n){
            
            if (tmp == 0){
                block_size = num/3;
                index = k-1;
                
                char order = 'a' + index/block_size; // 8/4=2
                ans = ans + order;
                index %= block_size; // 9%4 = 1
                // cout << order << " " << index << " " << endl;
            }
            else{

                block_size /= 2; // 4/2=2
                int prev = ans[tmp-1] - 'a';
                ans = ans + arr[prev][index/block_size]; //1/2=0
                index%=block_size; // 0%2=0
                // cout<< block_size << " "<<index << endl;
            }
            tmp++;
            
        }

        return ans;
        
    }
};
```
