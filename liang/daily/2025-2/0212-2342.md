## 2342. Max Sum of a Pair With Equal Sum of Digits
### C++
``` c++
class Solution {
public:
    int maximumSum(vector<int>& nums) {
        // digit sum range : 1~999999999(81)
        vector<int> dsum_maxVal(82,0);

        int tmp;
        int digit_sum=0;
        int max_val=-1;
        for (auto num : nums){
            tmp = num;
            digit_sum = 0;
            while(tmp > 0){
                digit_sum += tmp % 10;
                tmp /= 10;
            }
            if (dsum_maxVal[digit_sum-1]!=0)
                max_val = max(dsum_maxVal[digit_sum-1]+num,max_val);
            dsum_maxVal[digit_sum-1] = max(dsum_maxVal[digit_sum-1], num);
        }

        return max_val;
    }
};
```
