## 1726.Tuple with the same product
### C++
``` c++
class Solution {
public:
    int tupleSameProduct(vector<int>& nums) {
        if (nums.size()<4)
            return 0;
        
        int n = nums.size();
        unordered_map<int, int> product_map;
        int tmp;
        
        // fill the product table
        for (int i=0; i<n;++i){
            for (int j=i+1; j<n;++j){
                tmp = nums[i]*nums[j];
                product_map[tmp-1]++;
            }
        }

        // check the product table
        int ans = 0;
        for (const auto& [prod, freq] : product_map){
            if (freq > 1)
                ans += freq * (freq-1) * 4;
        }
        

        return ans;

    }
};
```
