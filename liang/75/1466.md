## 1466. Reorder Routes to Make All Paths Lead to the City Zero
### C++
``` c++
class Solution {
public:
    int minReorder(int n, vector<vector<int>>& connections) {
        
        int flip = 0;
        int cur_idx = 0; // find "0" in connections

        vector<bool> visited(n); // record if the node is visited.

        // direction array - tip: + means -> - means -<
        vector<vector<int>> c_list(n);
        for (auto &c : connections){
            c_list[c[0]].push_back(c[1]);
            c_list[c[1]].push_back(-c[0]);
        }
        
        dfs(cur_idx, c_list, flip, visited);

        return flip;
    }
private:
    void dfs(int cur_idx, vector<vector<int>> &c_list, int &flip, vector<bool> &visited){
        // mark visited true
        visited[cur_idx] = true;
        for (auto it : c_list[cur_idx]){
            if (!visited[abs(it)]){
                if (it>0){
                    flip++;
                    // cout << "it : " << it << endl;
                }
                    
                dfs(abs(it), c_list, flip, visited);
            }
        }
        return;
    }
};
```
