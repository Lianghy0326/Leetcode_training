class Solution {
public:
    string create_hash_vector(vector<int> &col){
       string hash_str= "";
       for (auto &c : col)
       {
        hash_str = hash_str + ('0'+to_string(c));
       }
       return hash_str;
    }
    int equalPairs(vector<vector<int>>& grid) {
        
        // iterate through grid and hash it. (to string)
        // string 的好處是不用存vector作為hash key
        
        // create hash O(N)
        unordered_map<string, int> map;
        for (auto &row : grid){
            map[create_hash_vector(row)]++;
            // for (auto &r:row){
            //     cout<<r;
            // }
            // cout<<endl;
        }
        int n = grid.size();
        vector<int> col(n); // save col vector
        int count;
        for (int i=0;i<n;++i){
            
            for(int j=0;j<n;++j){
                // cout << grid[j][i];
                col[j] = grid[j][i];
            }
            // cout<<endl;
            // check hash key num
            count += map[create_hash_vector(col)];
            
        }
        return count;
    }
};

class Solution {
public:
    int equalPairs(vector<vector<int>>& grid) {
        // Method 2 - use hashmap to store vector<int> as key and store count_val(int) as value
        // use find with condition if (it==hash_map.end()), if you access to the end of the key of hash map , 
        // return and seach for another vector( Check if the vector exists in hashmap, iterate through the rows)

        const int n = grid.size();
        // hashmap
        map<vector<int>,int> col_count;

        // iterate through grid and push the col into col_vec
        for (int i=0;i<n;++i){
            vector<int> col(n);
            for (int j=0;j<n;++j){
                
                col[j] = grid[j][i];
            }
            col_count[col]++;
        }

        // Iterate through rows and check if the row in hashmap.
        int count=0;
        for (auto &row : grid){
            // col_count[row] will trigger the map col_count to initialize "row" to 0(value).
            if (col_count[row]>0){
                count += col_count[row];
            }
        }
        return count;
    }
};

// hashmap + unordered_map : https://blog.csdn.net/happykocola/article/details/74188452

// C++ std will use this function to compute each hash value.
// C++ std-lib `unordered_map` must use operator to construct the hash-function of vector<int>.

// size_t : return type , usually unsign integer so use size_t
// operator() : 運算重載，可像函數一樣被調用
// (const vector<int>& vec) const : vector<int> as input and void copying it. 

// const 在方法結尾：const 在方法後面出現，表示這個方法不會修改該類的任何成員變數 (即 CustorMized_hash 這個類別)
// 沒 add tail const 會報錯 hash function must be invocable with an argument of key type 意思是 map hash function需是可以接受的類別
//  if( !CountRows.contains(row)) 會報錯因為 unordered_map 在使用 contains 時會調用 hashmap 觸發錯誤 
class CustorMized_hash {
    public:
        size_t operator() (const vector<int>& vec) const{
            // provide hash value
            size_t hash_value = vec.size(); // include size in hash
            for (int v : vec){
                // default hash function hash<type>{} (x)
                hash_value ^= std::hash<int>()(v) + 0x9e3779b9 + (hash_value << 6) + (hash_value >>2);
                // 0x9e3779b9 is the hash table 好數字
            }
            return hash_value;
        }
};

class Solution {
public:
    // Method - 3

    int equalPairs(vector<vector<int>>& grid) {
        // use .contain() to check whether the key exists in map or unordred_map
        
        // Get the transpose matrix grid^T
        const int n = grid.size();
        vector<vector<int>> grid_t(n,vector<int>(n));
        for (size_t i=0; i<n; ++i){
            for (size_t j=0;j<n;++j){
                grid_t[i][j] = grid[j][i];
            }
        }

        // Initialize hash map using unordred_map
        // unordred_map doesn't support "vector<int> as key", you need to custumize a hash-function.
        unordered_map<vector<int>,int, CustorMized_hash> CountRows;
        for (const vector<int>& row : grid){
            // use contain to check if the key exists.
            if( !CountRows.contains(row)){
                CountRows[row]=1;
            }
            else{
                CountRows[row]++;
            }
        }
        // init ans
        int ans = 0;
        // Iterate through rows in grid_t
        for (const vector<int>& row : grid_t){
            if (CountRows.contains(row)){
                ans += CountRows[row];
            }
        }

        return ans;
    }
};
